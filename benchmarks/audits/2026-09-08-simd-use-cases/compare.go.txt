package runtime

import (
	"encoding/binary"
	"math/bits"
	"simd/archsimd"
)

func auditSIMDCaseCompare(a, b string) int {
	limit := min(len(a), len(b))
	i := 0
	for ; i < min(4, limit); i++ {
		ca, cb := asciiLower(a[i]), asciiLower(b[i])
		if ca != cb {
			if ca < cb {
				return -1
			}
			return 1
		}
	}
	upperA := archsimd.BroadcastUint8x16('A')
	upperSpan := archsimd.BroadcastUint8x16('Z' - 'A')
	foldBit := archsimd.BroadcastUint8x16('a' - 'A')
	for ; limit-i >= 16; i += 16 {
		var at, bt [16]byte
		copy(at[:], a[i:i+16])
		copy(bt[:], b[i:i+16])
		av, bv := archsimd.LoadUint8x16Array(&at), archsimd.LoadUint8x16Array(&bt)
		am := av.Sub(upperA).LessEqual(upperSpan)
		bm := bv.Sub(upperA).LessEqual(upperSpan)
		av = av.Or(am.ToInt8x16().ToBits().And(foldBit))
		bv = bv.Or(bm.ToInt8x16().ToBits().And(foldBit))
		if av.NotEqual(bv).ToInt8x16().ToBits().ReduceMax() != 0 {
			for j := range 16 {
				ca, cb := asciiLower(a[i+j]), asciiLower(b[i+j])
				if ca != cb {
					if ca < cb {
						return -1
					}
					return 1
				}
			}
		}
	}
	for ; i < limit; i++ {
		ca, cb := asciiLower(a[i]), asciiLower(b[i])
		if ca != cb {
			if ca < cb {
				return -1
			}
			return 1
		}
	}
	return auditCompareLengths(a, b)
}

func auditSIMDCaseEqual(a, b string) bool {
	if len(a) != len(b) {
		return false
	}
	i := 0
	for ; i < min(4, len(a)); i++ {
		if asciiLower(a[i]) != asciiLower(b[i]) {
			return false
		}
	}
	upperA := archsimd.BroadcastUint8x16('A')
	upperSpan := archsimd.BroadcastUint8x16('Z' - 'A')
	foldBit := archsimd.BroadcastUint8x16('a' - 'A')
	for ; len(a)-i >= 16; i += 16 {
		var at, bt [16]byte
		copy(at[:], a[i:i+16])
		copy(bt[:], b[i:i+16])
		av, bv := archsimd.LoadUint8x16Array(&at), archsimd.LoadUint8x16Array(&bt)
		am := av.Sub(upperA).LessEqual(upperSpan)
		bm := bv.Sub(upperA).LessEqual(upperSpan)
		av = av.Or(am.ToInt8x16().ToBits().And(foldBit))
		bv = bv.Or(bm.ToInt8x16().ToBits().And(foldBit))
		if av.NotEqual(bv).ToInt8x16().ToBits().ReduceMax() != 0 {
			return false
		}
	}
	for ; i < len(a); i++ {
		if asciiLower(a[i]) != asciiLower(b[i]) {
			return false
		}
	}
	return true
}

// Seven-bit addition cannot carry into another byte. The two additions select
// A-Z using their high bits; the original high bits exclude non-ASCII bytes.
func auditGoFoldWord(word uint64) uint64 {
	low := word & 0x7f7f7f7f7f7f7f7f
	upper := (low + 0x3f3f3f3f3f3f3f3f) & ^(low + 0x2525252525252525) & ^word & 0x8080808080808080
	return word | upper>>2
}

func auditGoCaseCompare(a, b string) int {
	limit := min(len(a), len(b))
	i := 0
	for ; i < min(4, limit); i++ {
		ca, cb := asciiLower(a[i]), asciiLower(b[i])
		if ca != cb {
			if ca < cb {
				return -1
			}
			return 1
		}
	}
	for ; limit-i >= 8; i += 8 {
		av := auditGoFoldWord(binary.LittleEndian.Uint64([]byte(a[i : i+8])))
		bv := auditGoFoldWord(binary.LittleEndian.Uint64([]byte(b[i : i+8])))
		if av != bv {
			shift := bits.TrailingZeros64(av^bv) &^ 7
			if byte(av>>shift) < byte(bv>>shift) {
				return -1
			}
			return 1
		}
	}
	for ; i < limit; i++ {
		ca, cb := asciiLower(a[i]), asciiLower(b[i])
		if ca != cb {
			if ca < cb {
				return -1
			}
			return 1
		}
	}
	return auditCompareLengths(a, b)
}

func auditGoCaseEqual(a, b string) bool {
	if len(a) != len(b) {
		return false
	}
	i := 0
	for ; i < min(4, len(a)); i++ {
		if asciiLower(a[i]) != asciiLower(b[i]) {
			return false
		}
	}
	for ; len(a)-i >= 8; i += 8 {
		av := auditGoFoldWord(binary.LittleEndian.Uint64([]byte(a[i : i+8])))
		bv := auditGoFoldWord(binary.LittleEndian.Uint64([]byte(b[i : i+8])))
		if av != bv {
			return false
		}
	}
	for ; i < len(a); i++ {
		if asciiLower(a[i]) != asciiLower(b[i]) {
			return false
		}
	}
	return true
}

func auditCompareLengths(a, b string) int {
	switch {
	case len(a) < len(b):
		return -1
	case len(a) > len(b):
		return 1
	default:
		return 0
	}
}
