package runtime

import (
	"simd/archsimd"
	"strings"
)

const auditRegexpMetaChars = `\.+*?()|[]{}^$`

func auditGoRegexpQuotedSize(text string, limit int) (int, bool) {
	if len(text) > limit {
		return 0, false
	}
	size := len(text)
	for i := 0; i < len(text); {
		n := strings.IndexAny(text[i:], auditRegexpMetaChars)
		if n < 0 {
			break
		}
		if size == limit {
			return 0, false
		}
		size++
		i += n + 1
		// Rebuilding IndexAny's character set per adjacent metacharacter costs
		// more than the original loop, so dense runs keep scalar classification.
		if n < 16 {
			for ; i < len(text); i++ {
				if regexpMetaByte(text[i]) {
					if size == limit {
						return 0, false
					}
					size++
				}
			}
		}
	}
	return size, true
}

func auditGoWriteRegexpQuoted(out *strings.Builder, text string) {
	start := 0
	for i := 0; i < len(text); {
		n := strings.IndexAny(text[i:], auditRegexpMetaChars)
		if n < 0 {
			break
		}
		i += n
		out.WriteString(text[start:i])
		out.WriteByte('\\')
		start = i
		i++
		if n < 16 {
			for ; i < len(text); i++ {
				if regexpMetaByte(text[i]) {
					out.WriteString(text[start:i])
					out.WriteByte('\\')
					start = i
				}
			}
		}
	}
	out.WriteString(text[start:])
}

func auditSIMDRegexpQuotedSize(text string, limit int) (int, bool) {
	if len(text) > limit {
		return 0, false
	}
	size := len(text)
	i := 0
	if len(text) >= 32 {
		// The low-nibble table stores the permitted high-nibble bits. High
		// nibbles 8-15 map to zero, so arbitrary non-ASCII bytes remain literal.
		lowTable := [16]byte{0, 0, 0, 0, 4, 0, 0, 0, 4, 4, 4, 164, 160, 160, 36, 8}
		highTable := [16]byte{1, 2, 4, 8, 16, 32, 64, 128}
		low := archsimd.LoadUint8x16Array(&lowTable)
		high := archsimd.LoadUint8x16Array(&highTable)
		nibble := archsimd.BroadcastUint8x16(15)
		zero := archsimd.BroadcastUint8x16(0)
		one := archsimd.BroadcastUint8x16(1)
		for ; len(text)-i >= 16; i += 16 {
			var tile [16]byte
			copy(tile[:], text[i:i+16])
			v := archsimd.LoadUint8x16Array(&tile)
			matches := low.LookupOrZero(v.And(nibble)).And(high.LookupOrZero(v.ShiftAllRight(4)))
			count := int(matches.NotEqual(zero).ToInt8x16().ToBits().And(one).ReduceSum())
			if count > limit-size {
				return 0, false
			}
			size += count
		}
	}
	for ; i < len(text); i++ {
		if regexpMetaByte(text[i]) {
			if size == limit {
				return 0, false
			}
			size++
		}
	}
	return size, true
}

func auditSIMDWriteRegexpQuoted(out *strings.Builder, text string) {
	start := 0
	i := 0
	if len(text) >= 32 {
		lowTable := [16]byte{0, 0, 0, 0, 4, 0, 0, 0, 4, 4, 4, 164, 160, 160, 36, 8}
		highTable := [16]byte{1, 2, 4, 8, 16, 32, 64, 128}
		low := archsimd.LoadUint8x16Array(&lowTable)
		high := archsimd.LoadUint8x16Array(&highTable)
		nibble := archsimd.BroadcastUint8x16(15)
		for ; len(text)-i >= 16; i += 16 {
			var tile [16]byte
			copy(tile[:], text[i:i+16])
			v := archsimd.LoadUint8x16Array(&tile)
			matches := low.LookupOrZero(v.And(nibble)).And(high.LookupOrZero(v.ShiftAllRight(4)))
			if matches.ReduceMax() == 0 {
				continue
			}
			for j := range 16 {
				if regexpMetaByte(text[i+j]) {
					out.WriteString(text[start : i+j])
					out.WriteByte('\\')
					start = i + j
				}
			}
		}
	}
	for ; i < len(text); i++ {
		if regexpMetaByte(text[i]) {
			out.WriteString(text[start:i])
			out.WriteByte('\\')
			start = i
		}
	}
	out.WriteString(text[start:])
}
