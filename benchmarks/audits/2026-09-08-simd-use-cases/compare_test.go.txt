package runtime

import (
	"math/rand/v2"
	"strings"
	"testing"
)

func TestAuditCompare(t *testing.T) {
	check := func(t *testing.T, a, b string) {
		t.Helper()
		wantCompare := auditReferenceCaseCompare(a, b)
		wantEqual := auditReferenceCaseEqual(a, b)
		for _, implementation := range []struct {
			name    string
			compare func(string, string) int
			equal   func(string, string) bool
		}{
			{"go", auditGoCaseCompare, auditGoCaseEqual},
			{"simd", auditSIMDCaseCompare, auditSIMDCaseEqual},
		} {
			if got := implementation.compare(a, b); got != wantCompare {
				t.Fatalf("%s compare(%q, %q) = %d; want %d", implementation.name, a, b, got, wantCompare)
			}
			if got := implementation.equal(a, b); got != wantEqual {
				t.Fatalf("%s equal(%q, %q) = %t; want %t", implementation.name, a, b, got, wantEqual)
			}
		}
	}
	t.Run("all_byte_pairs", func(t *testing.T) {
		for a := range 256 {
			for b := range 256 {
				left, right := strings.Repeat(string(byte(a)), 48), strings.Repeat(string(byte(b)), 48)
				check(t, left, right)
				check(t, "AaBb"+left, "aAbB"+right)
			}
		}
	})
	t.Run("vector_boundaries", func(t *testing.T) {
		for _, n := range []int{0, 1, 3, 4, 7, 8, 15, 16, 17, 19, 20, 21, 31, 32, 33, 35, 36, 37, 63, 64, 65, 128, 257} {
			a := strings.Repeat("A", n)
			b := strings.Repeat("a", n)
			check(t, a, b)
			check(t, a, b+"a")
			check(t, a+"a", b)
			for pos := range n {
				for _, mismatch := range []byte{0, 'Z', '[', '\\', ']', '^', '_', '`', '{', 127, 128, 255} {
					changed := []byte(b)
					changed[pos] = mismatch
					check(t, a, string(changed))
					check(t, string(changed), a)
				}
			}
		}
	})
	t.Run("random", func(t *testing.T) {
		rng := rand.New(rand.NewPCG(42, 73))
		for range 2000 {
			n := rng.IntN(1025)
			a, b := make([]byte, n), make([]byte, n)
			for i := range n {
				a[i] = byte(rng.Uint32())
				b[i] = asciiLower(a[i])
			}
			check(t, string(a), string(b))
			if n != 0 {
				b[rng.IntN(n)] = byte(rng.Uint32())
				check(t, string(a), string(b))
			}
		}
	})
	t.Run("allocations", func(t *testing.T) {
		a := strings.Repeat("AbC[]_\xff", 512)
		b := strings.Repeat("aBc[]_\xff", 512)
		for _, implementation := range []struct {
			name    string
			compare func(string, string) int
			equal   func(string, string) bool
		}{
			{"go", auditGoCaseCompare, auditGoCaseEqual},
			{"simd", auditSIMDCaseCompare, auditSIMDCaseEqual},
		} {
			t.Run(implementation.name, func(t *testing.T) {
				if got := testing.AllocsPerRun(100, func() {
					if implementation.compare(a, b) != 0 || !implementation.equal(a, b) {
						t.Fatal("folded values differ")
					}
				}); got != 0 {
					t.Fatalf("comparison allocations = %v; want 0", got)
				}
			})
		}
	})
}
